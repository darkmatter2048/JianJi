# 以及拆成多块、各自带着缩放与过冲登场的标题。

aspect = 16:9
fps = 30

# ================================================================== 主角 ---

macbook = model:macbook-pro-13-2020 "MacBook Pro 13\"" color=#f5a524

# 0.00s 黑场里的一个小影子，从深处冲进来
@macbook t=0.00 p=[0.30,-0.30,-4.00] s=0 a=[0,-95,0] sc=[0.28,0.28,0.28] f=- v=1 scene=bg=media,bgmedia=底图开场,env=0.15,exp=0.3

# 1.70s 落位，灯全亮
@macbook t=1.70 p=[0,0,0] s=0 a=[0,-8,0] sc=[0.90,0.90,0.90] f=- v=1 scene=bg=media,bgmedia=底图开场,env=1.35,exp=1.15

# 2.30s 掀盖，背景同时切成彩色光斑
@macbook t=2.30 p=[0,0,0] s=70 a=[0.02,-6,0] f=壁纸 v=[0:0.3,0.5:1.9,1:0.7] scene=bg=media,bgmedia=底图彩色,env=1.4,exp=1.25

# 2.85s 开到底，稳
@macbook t=2.85 p=[0,0,0] s=112 a=[0.02,4,0] sc=[0.95,0.95,0.95] f=壁纸 v=1 scene=bg=media,bgmedia=底图彩色,env=1.25,exp=1.15

# 5.60s 微微仰一点，喘口气
@macbook t=5.60 p=[0,0,0.04] s=112 a=[0.03,6,0] f=壁纸 v=1

# 6.10s 屏幕切成代码，背景转冷
@macbook t=6.10 p=[0,0,0.04] s=112 a=[0.03,6,0] f=代码 v=[0:0.5,0.5:1.6,1:0.8] scene=bg=media,bgmedia=底图极速,env=1.1,exp=1.2

# 8.60s 360° 甩身，同时拉远，背景换成动态光
@macbook t=8.60 p=[0,0,-0.60] s=112 a=[0,-360,0] sc=[0.72,0.72,0.72] f=代码 v=[0:0.15,0.4:2.4,0.75:0.85,1:0.6] scene=bg=media,bgmedia=背景动画,env=1.5,exp=1.35

# 10.60s 转回正面落地，屏幕切进动态片段
@macbook t=10.60 p=[0,0,0] s=112 a=[0,0,0] sc=[1,1,1] f=开场 v=[0:0.4,0.5:1.5,1:0.7] scene=bg=media,bgmedia=背景动画,env=1.45,exp=1.3

# 12.40s 猛推近，屏幕是编辑器本体
@macbook t=12.40 p=[0,0,0.12] s=112 a=[0.05,22,0] sc=[1.18,1.18,1.18] f=界面 v=[0:0.2,0.5:2.2,1:0.75] scene=bg=media,bgmedia=底图极速,env=1.15,exp=1.1

# 15.60s 拉回，屏幕变片尾卡，灯光转暖收拢
@macbook t=15.60 p=[0,0,0] s=112 a=[0.02,-3,0] sc=[0.90,0.90,0.90] f=片尾 v=[0:0.3,0.5:1.8,1:0.7] scene=bg=media,bgmedia=底图片尾,env=0.95,exp=1

# 19.00s 定格收尾
@macbook t=19.00 p=[0,0,0] s=112 a=[0.02,-3,0] sc=[0.90,0.90,0.90] f=片尾 v=1

# 动态片段的附加轨道：先正常 1s，再慢放一小段（裁剪 + 变速）
@macbook.clip a m=开场 t=10.60 in=0.00 out=1.00 v=1
@macbook.clip b m=开场 t=11.60 in=2.60 out=3.40 v=[0:0.5,0.5:2,1:0.5]
